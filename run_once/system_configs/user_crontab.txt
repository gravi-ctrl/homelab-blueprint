# .------------ minute (0 - 59)
# | .---------- hour (0 - 23)

# Healthchecks.io Server/Internet monitor
*/5 * * * * cron-guard --mode mute "HC.io Ping" "curl -fsS --retry 3 '$SERVER_HC_URL'"

*/5 * * * * cron-guard --mode fail "NextDNS IP Update" "curl -fsS --retry 3 '$NEXTDNS_URL'"

# Pulls and deploys stack updates done by Renovate
*/15 * * * * cron-guard --mode fail "GitOps Deployer" "/opt/rabbit-hole/gitops-deploy.sh"

# Sync Personal and Work Obsidian vaults to Git
*/30 * * * * cron-guard --mode fail --repeat-after 1d --recovery "Obsidian Notes Sync" "/opt/rabbit-hole/git-auto-sync.py '$DATA_DIR/syncthing/Backup/obsidian-notes/personal' 'Obsidian Personal'; /opt/rabbit-hole/git-auto-sync.py '$DATA_DIR/syncthing/Backup/obsidian-notes/work' 'Obsidian Work'"

5 4 * * * cron-guard --mode fail "Pi-hole Gravity & Regex" "/opt/stacks/pihole/update-pihole-regex.sh"

# Export Nextcloud Calendars (.ics) and Contacts (.vcf)
20 4 * * * cron-guard --mode fail "Nextcloud CalCard Backup" "docker exec calcardbackup /opt/calcardbackup/calcardbackup /var/www/html -o /backup/syncthing/Backup/self-hosted/contacts-calendars_backup -s"

# Apply file renaming rules to Paperless-ngx documents
35 4 * * * cron-guard --mode fail "Paperless Auto Renamer" "docker exec -i paperless-ngx python3 manage.py document_renamer"

# Snapshot system state and build UI dashboards
0 5 * * * cron-guard --mode fail "Scripts & System Configs Backup" "/opt/rabbit-hole/backup-scripts-git.sh"

# Runs a Borgmatic backup
0 2 * * 0 cron-guard --mode all "Borg Backup" "borgmatic"

# cert-manager: regenerate & upload SSL certs to NPM
0 3 1 * * cron-guard --mode fail "cert-mgr: Renew SSL certs" "/opt/rabbit-hole/cert-manager/cert-manager.sh renew"
